import math

def resolver_equacao_segundo_grau(a, b, c):
    """
    Resolve uma equação do segundo grau no formato ax^2 + bx + c = 0.
    Retorna as raízes da equação.
    """

    if a == 0:
        raise ValueError("O coeficiente 'a' não pode ser 0 em uma equação do segundo grau.")

    # Calcula o discriminante (delta)
    delta = b**2 - 4*a*c

    if delta < 0:
        return "A equação não possui raízes reais."

    elif delta == 0:
        x = -b / (2*a)
        return (x,)  # raiz única

    else:
        x1 = (-b + math.sqrt(delta)) / (2*a)
        x2 = (-b - math.sqrt(delta)) / (2*a)
        return (x1, x2)


# Exemplo de uso
a = 1
b = -3
c = 2

resultado = resolver_equacao_segundo_grau(a, b, c)
print("Raízes:", resultado)
